cmake_minimum_required(VERSION 3.28)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(siny_ort_runtime LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "Static release CRT" FORCE)
foreach(flag ONNX_USE_MSVC_STATIC_RUNTIME protobuf_MSVC_STATIC_RUNTIME ABSL_MSVC_STATIC_RUNTIME)
  set(${flag} ON CACHE BOOL "" FORCE)
endforeach()
set(onnxruntime_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(onnxruntime_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ONNXRUNTIME_SOURCE_DIR}/cmake" runtime EXCLUDE_FROM_ALL)
# Traverse the configured target graph, including imported aliases and link-only edges.
# ORT patch lines rename dependency targets; stale hardcoded Abseil/test names are unsafe.
function(siny_collect_archive target)
  if(NOT TARGET "${target}")
    return()
  endif()
  get_target_property(alias "${target}" ALIASED_TARGET)
  if(alias)
    set(target "${alias}")
  endif()
  get_property(seen GLOBAL PROPERTY siny_seen_targets)
  if("${target}" IN_LIST seen)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY siny_seen_targets "${target}")
  get_target_property(kind "${target}" TYPE)
  if(kind STREQUAL "STATIC_LIBRARY")
    set_property(GLOBAL APPEND PROPERTY siny_archive_targets "${target}")
    set_property(GLOBAL APPEND PROPERTY siny_archive_inputs "$<TARGET_FILE:${target}>")
  endif()
  foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(edges "${target}" "${property}")
    if(edges)
      foreach(edge IN LISTS edges)
        if(TARGET "${edge}")
          siny_collect_archive("${edge}")
        else()
          # Extract target identifiers inside LINK_ONLY / BUILD_INTERFACE expressions.
          string(REGEX MATCHALL "[A-Za-z_][A-Za-z0-9_:.+-]*" names "${edge}")
          foreach(name IN LISTS names)
            if(TARGET "${name}")
              siny_collect_archive("${name}")
            endif()
          endforeach()
        endif()
      endforeach()
    endif()
  endforeach()
endfunction()
# The public interface owns ONNX/protobuf and the other external link edges.
# Component archive targets alone carry include dependencies, not all link edges.
if(NOT TARGET onnxruntime)
  message(FATAL_ERROR "Required public ORT interface missing")
endif()
siny_collect_archive(onnxruntime)
foreach(component onnxruntime_session onnxruntime_optimizer onnxruntime_providers
    onnxruntime_util onnxruntime_framework onnxruntime_graph onnxruntime_mlas
    onnxruntime_common onnxruntime_flatbuffers)
  if(NOT TARGET "${component}")
    message(FATAL_ERROR "Required ORT component missing: ${component}")
  endif()
  siny_collect_archive("${component}")
endforeach()
get_property(archive_targets GLOBAL PROPERTY siny_archive_targets)
get_property(archive_inputs GLOBAL PROPERTY siny_archive_inputs)
foreach(required onnx onnx_proto)
  if(NOT "${required}" IN_LIST archive_targets)
    message(FATAL_ERROR "Missing ONNX archive in the public dependency graph: ${required}")
  endif()
endforeach()
list(LENGTH archive_targets archive_count)
if(archive_count LESS 9)
  message(FATAL_ERROR "Incomplete native dependency graph")
endif()
message(STATUS "Bundling ${archive_count} actual static dependency targets: ${archive_targets}")
set(bundled_archive "${CMAKE_CURRENT_BINARY_DIR}/onnxruntime.lib")
add_custom_command(OUTPUT "${bundled_archive}"
  COMMAND "${CMAKE_AR}" /NOLOGO "/OUT:${bundled_archive}" ${archive_inputs}
  DEPENDS ${archive_targets} COMMAND_EXPAND_LISTS VERBATIM)
add_custom_target(siny_runtime ALL DEPENDS "${bundled_archive}")
install(FILES "${bundled_archive}" DESTINATION lib)
install(DIRECTORY "${ONNXRUNTIME_SOURCE_DIR}/include/onnxruntime/core/session/"
  DESTINATION include FILES_MATCHING PATTERN "*.h")
